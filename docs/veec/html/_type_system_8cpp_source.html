38ConversionRank TypeSystem::rankConversion(Type* from, Type* to, ConversionMode mode) const {
39 VEE_ASSERT(from != nullptr, "from type is null");
40 VEE_ASSERT(to != nullptr, "to type is null");
41
42 // Identical types -> exact match
43 if (from == to) {
44 return ConversionRank::ExactMatch;
45 }
46
47 // Lookup in conversion rules
48 auto it = _conversionRules.find({from, to});
49 if (it == _conversionRules.end()) {
50 return ConversionRank::NoConversion;
51 }
52
53 ConversionRank rank = it->second.rank;
54
55 // Ignore explicits if implicit mode
56 if (mode == ConversionMode::Implicit && rank == ConversionRank::ExplicitConversion) {
57 return ConversionRank::NoConversion;
58 }
59
60 return rank;
61}
62u32 TypeSystem::conversionCost(Type* from, Type* to, ConversionMode mode) const {
63 ConversionRank rank = rankConversion(from, to, mode);
64
65 switch (rank) {
66 case ConversionRank::ExactMatch:
67 return 0;
68 case ConversionRank::Promotion:
69 return 1;
70 case ConversionRank::Conversion:
71 return 2;
72 case ConversionRank::NarrowingConversion:
73 return 3;
74 case ConversionRank::UserDefinedConversion:
75 return 4;
76 case ConversionRank::ExplicitConversion:
77 if (mode == ConversionMode::Explicit) {
78 return 5;

82 case ConversionRank::NoConversion:
83 return kNoConversionCost;
