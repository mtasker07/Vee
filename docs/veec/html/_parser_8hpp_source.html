286 inline const Token* expect(TokenType expectedType, const diagnostics::DiagnosticDescriptor<ArgCount>& diagDesc, Args&&... args) {
287 if (match(expectedType)) {

293 return nullptr;

299 inline void synchronize() {
300 while (!isAtEnd()) {
301 const Token& tk = peek();
302
303 if (tk.type() == TokenType::SemiColon) {
304 advance();
305 return;
306 }
307 else if (tk.isKeyword()) {
308 return;
309 }
310
311 advance();
312 }
313 }
317 inline void skipTrivia() {
318 while (_currentTokenIndex < static_cast<int>(_tokenList.tokens.size())
319 && _tokenList.tokens[_currentTokenIndex].type() != TokenType::EndOfFile
320 && _tokenList.tokens[_currentTokenIndex].isTrivia()) {
321 _currentTokenIndex++;
