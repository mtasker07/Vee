70ast::CompilationUnitNode* Parser::parseCompilationUnit() {
71 // Cant parse if there are no tokens
72 if (_tokenList.tokens.empty()) {
73 return nullptr;
74 }
75
76 std::vector<ast::ItemNode*> items;
77
78 while (!isAtEnd()) {
79 // Parse a top-level item
80 ast::ItemNode* node = parseItem();
81 if (node) {
82 items.push_back(node);
83 }
84 else synchronize();
85 }
86
87 // If no statements were parsed, return nullptr to indicate an empty module.
88 // This also avoids using statements.front() and back() which throw
89 // on an empty vector
90 if (items.empty()) {
91 return nullptr;
92 }
93
94 source::SourceRange moduleRange = combineRanges(
95 items.front()->getRange(),
96 items.back()->getRange()
97 );
98 ast::CompilationUnitNode* compilationUnit = _ast.makeNode<ast::CompilationUnitNode>(
99 moduleRange,
100 std::move(items)
101 );
102 _ast.modules.push_back(compilationUnit);
103 return compilationUnit;
104}

124ast::QualifiedNameNode* Parser::parseQualifiedName() {
125 std::vector<ast::QualifiedNameSegment> segments;
126
127 const Token& beginToken = peek();
128
129 while (!isAtEnd()) {
130 ast::QualifiedNameSegment segment;
131
132 // Identifier
133 if (!match(TokenType::Identifier)) {
134 _ctx.diagnostics.report(diagnostics::ERROR_IDENTIFIER_EXPECTED, peek().range());
135 return nullptr;
136 }
137 const Token& idToken = previous();
138 source::Identifier idName = tokenToIdentifier(idToken);
139 segment.identifier = idName;
140
141 // '::' Indicates either more segments or generic args
142 if (!match(TokenType::ColonColon)) {
143 // No more segments or generic args
144 segments.push_back(segment);
145 break;
146 }
147
148 if (!check(TokenType::Less)) {
149 segments.push_back(segment);
150 continue; // More segments, no generic args for this segment
151 }
152
153 // Generic args for *this* segment
154 ast::GenericArgsNode* genericArgs = parseGenericArgs();
155 segment.genericArgs = genericArgs;
156 segments.push_back(segment);
157
158 // Do we have more segments?
159 if (!match(TokenType::ColonColon)) {
160 break;
161 }
162 }
163
164 const Token& endToken = previous();
165 const source::SourceRange nameRange = tokenRange(beginToken, endToken);
166 return _ast.makeNode<ast::QualifiedNameNode>(
167 nameRange, // range
168 std::move(segments) // segments
169 );
170}

172ast::GenericArgsNode* Parser::parseGenericArgs() {
173 VEE_ASSERT(match(TokenType::Less), "Expected '<' token before parsing generic arguments");
174 const Token& argsBegin = previous();
175
176 std::vector<ast::TypeNode*> args;
177 do {
178 ast::TypeNode* arg = parseType();
179 if (!arg) {
180 _ctx.diagnostics.report(diagnostics::ERROR_TYPE_EXPECTED, peek().range());
181 return nullptr;
182 }
183 args.push_back(arg);
184 } while (!isAtEnd() && match(TokenType::Comma));
185
186 expect(TokenType::Greater, diagnostics::ERROR_GREATER_EXPECTED);
187
188 const Token& argsEnd = previous();
189 source::SourceRange argsRange = tokenRange(argsBegin, argsEnd);
190 return _ast.makeNode<ast::GenericArgsNode>(
191 argsRange,
192 std::move(args)
193 );

242ast::ModuleDeclNode* Parser::parseModuleDeclaration() {
243 // Syntax:
244 // module <identifier> {
245 // <items>
246 // }
247
248 VEE_ASSERT(match(TokenType::Module), "Expected 'module' keyword before parsing module declaration");
249 const Token& moduleToken = previous();
250
251 const Token* idToken = expect(TokenType::Identifier, diagnostics::ERROR_IDENTIFIER_EXPECTED);
252 if (!idToken) {
253 return nullptr;
254 }
255
256 expect(TokenType::LCurly, diagnostics::ERROR_LCURLY_EXPECTED);
257
258 std::vector<ast::ItemNode*> items;
259 while (!isAtEnd() && !check(TokenType::RCurly)) {
260 ast::ItemNode* item = parseItem();
261 if (item) {
262 items.push_back(item);
263 }
264 }
265
266 expect(TokenType::RCurly, diagnostics::ERROR_RCURLY_EXPECTED);
267
268 source::SourceRange moduleRange = tokenRange(moduleToken, previous());
269 source::Identifier moduleName = tokenToIdentifier(*idToken);
270 return _ast.makeNode<ast::ModuleDeclNode>(
271 moduleRange, // range
272 moduleName, // name
273 std::move(items) // items
274 );

276ast::FunctionDeclNode* Parser::parseFunctionDeclaration() {
277 // Syntax:
278 // func <identifier>(<parameters>) [-> <return_type>] {
279 // <body>
280 // }
281
282 VEE_ASSERT(match(TokenType::Func), "Expected 'func' keyword before parsing function declaration");
283 const Token& funcToken = previous();
284
285 const Token* idToken = expect(TokenType::Identifier, diagnostics::ERROR_IDENTIFIER_EXPECTED);
286 if (!idToken) {
287 return nullptr;
288 }
289
290 expect(TokenType::LParen, diagnostics::ERROR_LPAREN_EXPECTED);
291
292 std::vector<ast::ParameterDeclNode*> parameters;
293 while (!isAtEnd() && !check(TokenType::RParen)) {
294 do {
295 ast::ParameterDeclNode* param = parseParameterDeclaration();
296 if (!param) {
297 return nullptr;
298 }
299 parameters.push_back(param);
300 } while (match(TokenType::Comma));
301 }
302
303 expect(TokenType::RParen, diagnostics::ERROR_RPAREN_EXPECTED);
304
305 ast::TypeNode* returnType = nullptr;
306 if (match(TokenType::RArrow)) {
307 returnType = parseType();
308 if (!returnType) {
309 return nullptr;
310 }
311 }
312
313 ast::BlockStmtNode* body = parseBlockStatement();
314 if (!body) {
315 return nullptr;
316 }
317
318 // TODO: Maybe previous() usage is bad here?
319 source::SourceRange funcRange = tokenRange(funcToken, previous());
320 source::Identifier funcName = tokenToIdentifier(*idToken);
321 return _ast.makeNode<ast::FunctionDeclNode>(
322 funcRange, // range
323 funcName, // name
324 std::move(parameters), // parameters
325 returnType, // return type
326 body // body
327 );
328}

453ast::VariableDeclNode* Parser::parseVariableDeclaration() {
454 ast::VariableDeclKind kind;
455 if (match(TokenType::Var))
456 kind = ast::VariableDeclKind::Var;
457 else if (match(TokenType::Let))
458 kind = ast::VariableDeclKind::Let;
459 else
460 VEE_FATAL("Expected 'var' or 'let' keyword before parsing variable declaration");
461
462 const Token& varToken = previous();
463
464 const Token* identifier = expect(TokenType::Identifier, diagnostics::ERROR_IDENTIFIER_EXPECTED);
465 if (!identifier) {
466 return nullptr;
467 }
468
469 // Explicit type
470 ast::TypeNode* type = nullptr;
471 if (match(TokenType::Colon)) {
472 type = parseType();
473 if (!type) {
474 return nullptr;
475 }
476 }
477
478 // Optional initializer
479 ast::ExpressionNode* initializer = nullptr;
480 if (match(TokenType::Equal)) {
481 initializer = parseExpression();
482 if (!initializer) {
483 return nullptr;
484 }
485 }
486
487 expect(TokenType::SemiColon, diagnostics::ERROR_SEMICOLON_EXPECTED);
488
489 source::SourceRange varRange = tokenRange(varToken, previous());
490 source::Identifier varName = tokenToIdentifier(*identifier);
491 return _ast.makeNode<ast::VariableDeclNode>(
492 varRange, // range
493 kind, // kind
494 varName, // name
495 type, // type
496 initializer // initializer
497 );
498}

581ast::IfStmtNode* Parser::parseIfStatement() {
582 // Syntax:
583 // if <condition> <then_branch> [else <else_branch>]
584
585 VEE_ASSERT(match(TokenType::If), "Expected 'if' keyword before parsing if statement");
586 const Token& ifToken = previous();
587
588 ast::ExpressionNode* condition = parseExpression();
589 if (!condition) {
590 _ctx.diagnostics.report(diagnostics::ERROR_EXPRESSION_EXPECTED, peek().range());
591 return nullptr;
592 }
593
594 ast::StatementNode* thenBranch = parseStatement();
595 if (!thenBranch) {
596 _ctx.diagnostics.report(diagnostics::ERROR_STATEMENT_EXPECTED, peek().range());
597 return nullptr;
598 }
599
600 ast::StatementNode* elseBranch = nullptr;
601 if (match(TokenType::Else)) {
602 elseBranch = parseStatement();
603 if (!elseBranch) {
604 _ctx.diagnostics.report(diagnostics::ERROR_STATEMENT_EXPECTED, peek().range());
605 return nullptr;
606 }
607 }
608
609 source::SourceRange ifRange = tokenRange(ifToken, previous());
610 return _ast.makeNode<ast::IfStmtNode>(
611 ifRange, // range
612 condition, // condition
613 thenBranch, // then branch
614 elseBranch // else branch
615 );
616}

665

667 const Token& forToken = previous();

689 }

738ast::UseStmtNode* Parser::parseUseStatement() {
739 VEE_ASSERT(match(TokenType::Use), "Expected 'use' keyword before parsing use statement");
740 const Token& useToken = previous();
741
742 std::vector<source::Identifier> modulePath;
743 std::vector<source::Identifier> imports;
744 while (!isAtEnd()) {
745 // MODULE PATH
746 if (!match(TokenType::Identifier)) {
747 _ctx.diagnostics.report(diagnostics::ERROR_MODULE_NAME_EXPECTED, peek().range());
748 return nullptr;
749 }
750 modulePath.push_back(tokenToIdentifier(previous()));
751
752 // identifier::
753 if (!match(TokenType::ColonColon)) {
754 break;
755 }
756
757 // identifier::{
758 if (!match(TokenType::LCurly)) {
759 continue;
760 }
761
762 // IMPORTS
763 do {
764 if (!match(TokenType::Identifier)) {
765 _ctx.diagnostics.report(diagnostics::ERROR_IMPORT_NAME_EXPECTED, peek().range());
766 return nullptr;
767 }
768 imports.push_back(tokenToIdentifier(previous()));
769 } while (!isAtEnd() && match(TokenType::Comma));
770 expect(TokenType::RCurly, diagnostics::ERROR_RCURLY_EXPECTED);
771 break;
772 }
773
774 expect(TokenType::SemiColon, diagnostics::ERROR_SEMICOLON_EXPECTED);
775
776 const Token& endToken = previous();

788ast::ExpressionNode* Parser::parseExpressionBP(i32 minBP) {
789 ast::ExpressionNode* lhs = parsePrefixExpression();
790 if (!lhs) {
791 return nullptr;
792 }
793
794 while (true) {
795 const Token& opToken = peek();
796 auto bp = getInfixBindingPower(opToken.type());
797
798 if (!bp || bp->left < minBP) {
799 break;
800 }
801
802 lhs = parseInfixExpression(lhs);
803 if (!lhs) {
804 return nullptr;
805 }
806 }
807
808 return lhs;

810ast::ExpressionNode* Parser::parsePrefixExpression() {
811 using TT = TokenType;
812

814 if (prefixBP) {

821
822 case TT::IntegerLiteral:
823 return parseIntLiteralExpression();
824 case TT::FloatLiteral:
825 return parseFloatLiteralExpression();
826 case TT::StringLiteral:
827 return parseStringLiteralExpression();
828 case TT::True:
829 case TT::False:
830 return parseBoolLiteralExpression();
831
832 case TT::LParen:
833 return parseParenthesizedExpression();
834
835 default:
836 _ctx.diagnostics.report(diagnostics::ERROR_EXPRESSION_EXPECTED, peek().range());
837 return nullptr;
838 }
