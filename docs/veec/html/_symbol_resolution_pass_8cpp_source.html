70void SymbolResolutionPass::resolveQualifiedName(ast::QualifiedNameNode& node) {
71 using SymKind = symbols::SymbolKind;
72
73 // Identify start scope for resolution
74 source::Identifier firstSegment = node.getSegments().front().identifier;
75 sema::Scope* s = _currentScope;
76 while (s != nullptr) {
77 if (lookupNameInScope(firstSegment.id, s) != nullptr) {
78 break;
79 }
80 s = s->getParent();
81 }
82
83 // Unknown first segment
84 if (s == nullptr) {
85 _ctx.diagnostics.report(
86 diagnostics::ERROR_UNKNOWN_IDENTIFIER,
87 node.getRange(),
88 _ctx.strings.get(firstSegment.id)
89 );
90 return;
91 }
92
93 // Validate each segment from initial scope
94 const auto& segments = node.getSegments();
95 symbols::Symbol* lastResolvedSymbol = nullptr;
96 for (size_t i = 0; i < segments.size(); ++i) {
97 basic::StringId id = segments[i].identifier.id;
98 symbols::Symbol* resolved = lookupNameInScope(id, s);
99 if (resolved == nullptr) {
100 // Unknown symbol (not first segment)
101 _ctx.diagnostics.report(
102 diagnostics::ERROR_NO_MEMBER_IN_SYMBOL,
103 node.getRange(),
104 symbols::Symbol::kindString(lastResolvedSymbol->getKind()),
105 _ctx.strings.get(segments[i - 1].identifier.id),
106 // ^^ Always safe since first segment is guaranteed to be valid
107 _ctx.strings.get(id)
108 );
109 return;
110 }
111 lastResolvedSymbol = resolved;
112
113 if (i == segments.size() - 1) {
114 // No more segments, break early to avoid checking for scope owner
115 break;
116 }
117
118 symbols::ScopeOwnerSymbol* sos = resolved->as<symbols::ScopeOwnerSymbol>();
119 if (!sos) {
120 std::string_view kindStr =
121 symbols::Symbol::kindString(resolved->getKind());
122
123 _ctx.diagnostics.report(
124 diagnostics::ERROR_SYMBOL_CANNOT_BE_QUALIFIED,
125 node.getRange(),
126 kindStr,
127 _ctx.strings.get(id)
128 );
129 return;
130 }
131
132 s = sos->getScope();
133 }
134 VEE_ASSERT(lastResolvedSymbol != nullptr, "Failed to resolve qualified name. Node:\n{}", node.toString(_ctx));
135 node.resolvedSymbol = lastResolvedSymbol;
136}

138 VEE_ASSERT(scope != nullptr, "Scope is null while looking up name: {}", _ctx.strings.get(nameId));
