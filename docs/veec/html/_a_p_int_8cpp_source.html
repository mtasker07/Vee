379APInt APInt::subtractUnsigned(const APInt& lhs, const APInt& rhs) {
380 VEE_ASSERT(lhs._bitWidth == rhs._bitWidth, "Bit widths must match");
381
382 size_t maxSize = std::max(lhs._words.size(), rhs._words.size());
383 std::vector<u32> result;
384 result.reserve(maxSize);
385
386 u64 borrow = 0;
387 for (size_t i = 0; i < maxSize; ++i) {
388 u64 lword = i < lhs._words.size() ? lhs._words[i] : 0;
389 u64 rword = i < rhs._words.size() ? rhs._words[i] : 0;
390
391 u64 subtrahend = rword + borrow;
392 if (lword < subtrahend) {
393 result.push_back(static_cast<u32>((1ULL << 32) + lword - subtrahend));
394 borrow = 1;
395 } else {
396 result.push_back(static_cast<u32>(lword - subtrahend));
397 borrow = 0;
398 }
399 }
400
401 APInt diff(lhs._bitWidth, std::move(result));
402 diff.applyWidthMask();
403 return diff;

405APInt APInt::multiplyUnsigned(const APInt& lhs, const APInt& rhs) {
406 VEE_ASSERT(lhs._bitWidth == rhs._bitWidth, "Bit widths must match");
407
408 if (lhs.isZero() || rhs.isZero()) {
409 return APInt::fromU64(lhs._bitWidth, 0);
410 }
411
412 std::vector<u32> result(lhs._words.size() + rhs._words.size(), 0);
413
414 for (size_t i = 0; i < lhs._words.size(); ++i) {
415 u64 carry = 0;
416 for (size_t j = 0; j < rhs._words.size(); ++j) {
417 u64 product = static_cast<u64>(lhs._words[i]) * rhs._words[j];
418 u64 sum = static_cast<u64>(result[i + j]) + product + carry;
419
420 result[i + j] = static_cast<u32>(sum);
421 carry = sum >> 32;
422 }
423
424 // Drop carry past top limb; fixed width wraps
425 if (i + rhs._words.size() < result.size()) {
426 result[i + rhs._words.size()] += static_cast<u32>(carry);
427 }
428 }
429
430 APInt product(lhs._bitWidth, std::move(result));
431 product.applyWidthMask();
432 return product;
433}
434APInt APInt::divideUnsigned(const APInt& lhs, const APInt& rhs) {
435 VEE_ASSERT(lhs._bitWidth == rhs._bitWidth, "Bit widths must match");
436
437 if (rhs.isZero()) {
438 VEE_FATAL("Division by zero");
439 }
440
441 if (compareUnsigned(lhs, rhs) < 0) {
442 return APInt::fromU64(lhs._bitWidth, 0);
443 }
444
445 if (lhs._bitWidth == 0) {
446 return APInt::fromU64(0, 0);
447 }
448
449 u32 extBitWidth = lhs._bitWidth + 1;
450 APInt divisor(extBitWidth, rhs._words);
451 divisor.applyWidthMask();
452
453 APInt remainder(extBitWidth, std::vector<u32>(limbCount(extBitWidth), 0));
454 APInt quotient(lhs._bitWidth, std::vector<u32>(limbCount(lhs._bitWidth), 0));
455
456 for (u32 bit = lhs._bitWidth; bit > 0; --bit) {
457 u32 carry = 0;
458 for (size_t i = 0; i < remainder._words.size(); ++i) {
459 u32 nextCarry = remainder._words[i] >> 31;
460 remainder._words[i] = (remainder._words[i] << 1) | carry;
461 carry = nextCarry;
462 }
463 remainder.applyWidthMask();
464
465 u32 sourceBit = bit - 1;
466 u32 sourceWord = sourceBit / 32;
467 u32 sourceShift = sourceBit % 32;
468 if (sourceWord < lhs._words.size() && ((lhs._words[sourceWord] >> sourceShift) & 1u) != 0) {
469 remainder._words[0] |= 1u;
470 }
471
472 if (compareUnsigned(remainder, divisor) >= 0) {
473 remainder = subtractUnsigned(remainder, divisor);
474
475 u32 quotientWord = sourceBit / 32;
476 u32 quotientShift = sourceBit % 32;
477 quotient._words[quotientWord] |= (1u << quotientShift);
478 }
479 }
480
481 quotient.applyWidthMask();
482 return quotient;
483}
484int APInt::compareUnsigned(const APInt& lhs, const APInt& rhs) {
485 VEE_ASSERT(lhs._bitWidth == rhs._bitWidth, "Bit widths must match");
486
487 size_t lhsSize = trimmedSize(lhs._words);
488 size_t rhsSize = trimmedSize(rhs._words);
489
490 if (lhsSize != rhsSize) {
491 return lhsSize < rhsSize ? -1 : 1;
492 }
493
494 for (size_t i = lhsSize; i > 0; --i) {
495 size_t index = i - 1;
496 if (lhs._words[index] != rhs._words[index]) {
497 return lhs._words[index] < rhs._words[index] ? -1 : 1;
498 }
499 }

501 return 0;
502}
