27basic::Result<SourceFileId, SourceLoadError> SourceManager::loadFile(const fs::Path& path) {
28 // Check if the file is already loaded
29 auto it = _fileMap.find(path);
30 if (it != _fileMap.end()) {
31 return it->second;
32 }
33
34 // Load the file contents
35 // TODO: abstract into file system class/subsystem
36 std::string contents;
37 try {
38 std::ifstream file(path.str().data());
39 if (!file) {
40 SourceLoadError error;
41 error.kind = SourceLoadErrorKind::FileNotFound;
42 error.filePath = path;
43 return error;
44 }
45 contents.assign((std::istreambuf_iterator<char>(file)), std::istreambuf_iterator<char>());
46 } catch (const std::exception&) {
47 SourceLoadError error;
48 error.kind = SourceLoadErrorKind::UnknownError;
49 error.filePath = path;
50 return error;
51 }
52
53 // Create a new SourceFile and store it
54 SourceFileId fileId = static_cast<SourceFileId>(_files.size());
55 SourceFile source = SourceFile(fileId, path, contents);
56 _files.push_back(std::move(source));
57 _fileMap[path] = fileId;
58
59 return fileId;
60}
