72void TypeCheckerPass::visitIntLiteralExpr(ast::IntLiteralExprNode& node) {
73 using BTK = types::BuiltinTypeKind;
74
75 ast::AstWalker::visitIntLiteralExpr(node);
76
77 // Integer literals are always i32
78 // TODO: Support suffixes for different literal types
79 types::BuiltinType* intType = _ctx.types.table.getBuiltin(types::BuiltinTypeKind::I32);
80 _ctx.types.table.setNodeType(&node, intType);
81
82 // Validate size of integer literal
83 const basic::BigInt& value = node.getValue();
84
85 u32 bitWidth = 0;
86 bool isSigned = false;
87 BTK builtinKind = intType->getBuiltinKind();
88 switch (builtinKind) {
89 case BTK::I8:
90 case BTK::U8:
91 bitWidth = 8;
92 isSigned = (builtinKind == BTK::I8);
93 break;
94 case BTK::I16:
95 case BTK::U16:
96 bitWidth = 16;
97 isSigned = (builtinKind == BTK::I16);
98 break;
99 case BTK::I32:
100 case BTK::U32:
101 bitWidth = 32;
102 isSigned = (builtinKind == BTK::I32);
103 break;
104 case BTK::I64:
105 case BTK::U64:
106 bitWidth = 64;
107 isSigned = (builtinKind == BTK::I64);
108 break;
109 }
110
111 if (isSigned) {
112 if (!value.fitsInSigned(bitWidth)) {
113 _ctx.diagnostics.report(
114 diagnostics::ERROR_INT_LITERAL_OUT_OF_RANGE,
115 node.getRange(),
116 value.toString(),
117 bitWidth
118 );
119 }
120 } else {
121 if (!value.fitsInUnsigned(bitWidth)) {
122 _ctx.diagnostics.report(
123 diagnostics::ERROR_UINT_LITERAL_OUT_OF_RANGE,
124 node.getRange(),
125 value.toString(),
126 bitWidth
127 );
128 }
129 }

155void TypeCheckerPass::visitUnaryExpr(ast::UnaryExprNode& node) {
156 ast::AstWalker::visitUnaryExpr(node);
157 // ^^ Sets operand type
158
159 types::ErrorType* errorType = _ctx.types.table.getError();
160 types::Type* operandType = _ctx.types.table.getNodeType(node.getOperand());
161 VEE_ASSERT(operandType != nullptr, "Failed to infer type for operand of unary expression");
162
163 // Propogate errors early
164 if (operandType == errorType) {
165 _ctx.types.table.setNodeType(&node, errorType);
166 return;
167 }
168
169 types::Type* resultType = errorType;
170
171 // Lookup operator
172 symbols::UnaryOperatorKind semaOp = astToSemaUnaryOp(node.getOperator());
173 std::vector<symbols::OperatorSymbol*> result = lookupUnaryOperator(semaOp, operandType);
174
175 if (result.empty()) {
176 // No matching operator found
177 _ctx.diagnostics.report(
178 diagnostics::ERROR_UNARY_OPERATOR_NOT_FOUND,
179 node.getRange(),
180 ast::toString(node.getOperator()),
181 operandType->toString()
182 );
183 } else if (result.size() > 1) {
184 // Ambiguous operator
185 _ctx.diagnostics.report(
186 diagnostics::ERROR_UNARY_OPERATOR_AMBIGUOUS,
187 node.getRange(),
188 ast::toString(node.getOperator()),
189 operandType->toString()
190 );
191 } else {
192 // Found exactly one matching operator
193 symbols::OperatorSymbol* opSymbol = result[0];
194 node.symbol = opSymbol;
195 resultType = opSymbol->getResultType();
196
197 // Emit conversion diagnostics
198 emitImplicitConversionDiagnostics(
199 node.getRange(),
200 operandType,
201 opSymbol->getOperandType(0)
202 );
203 }
204
205 _ctx.types.table.setNodeType(&node, resultType);
206}

230

232 // No matching operator found
233 _ctx.diagnostics.report(
234 diagnostics::ERROR_BINARY_OPERATOR_NOT_FOUND,
235 node.getRange(),
236 ast::toString(node.getOperator()),
237 lType->toString(),
238 rType->toString()
239 );
240 } else if (result.size() > 1) {
241 // Ambiguous operator
242 _ctx.diagnostics.report(
243 diagnostics::ERROR_BINARY_OPERATOR_AMBIGUOUS,
244 node.getRange(),
245 ast::toString(node.getOperator()),
246 lType->toString(),
247 rType->toString()
248 );
249 } else {
250 // Found exactly one matching operator
251 symbols::OperatorSymbol* opSymbol = result[0];
252 node.symbol = opSymbol;
253 resultType = opSymbol->getResultType();
254
255 // Emit conversion diagnostics
256 emitImplicitConversionDiagnostics(
257 node.getRange(),
258 lType,

292void TypeCheckerPass::visitNameExpr(ast::NameExprNode& node) {
293 ast::AstWalker::visitNameExpr(node);
294
295 types::ErrorType* errorType = _ctx.types.table.getError();
296
297 // Type is just type of resolved symbol
298 const symbols::Symbol* resolvedSymbol = node.getResolvedSymbol();
299 if (resolvedSymbol) {
300 types::Type* symbolType = errorType;
301
302 // Function sets are a special case, since we dont actually know which overload
303 // is being called in the current context. This is the one case where we can dont
304 // set the type of the symbol (keep it null), and we will instead rely on CallExpr to set it
305 if (auto* funcSet = resolvedSymbol->as<symbols::FunctionSetSymbol>()) {
306 return;
307 }
308 // Not a function set -> standard symbol resolution
309 else switch (resolvedSymbol->getKind()) {
310 case symbols::SymbolKind::Variable:
311 symbolType = resolvedSymbol->as<symbols::VariableSymbol>()->getType();
312 break;
313 case symbols::SymbolKind::Function:
314 // NOTE: Technically, this should never run, since we always bind
315 // names to function set symbols. However, just in case this slips through
316 // somewhere this logic is still reasonable and wont break anything.
317 symbolType = resolvedSymbol->as<symbols::FunctionSymbol>()->getType();
318 break;
319 case symbols::SymbolKind::Field:
320 symbolType = resolvedSymbol->as<symbols::FieldSymbol>()->getType();
321 break;
322
323 default:
324 // Not a type-bearing symbol
325 break;
326 }
327 _ctx.types.table.setNodeType(&node, symbolType);
328 } else {
329 // Cannot infer unresolved symbol
330 _ctx.types.table.setNodeType(&node, errorType);
331 }

333void TypeCheckerPass::visitCallExpr(ast::CallExprNode& node) {
334 ast::AstWalker::visitCallExpr(node);
335
336 types::ErrorType* errorType = _ctx.types.table.getError();
337
338 // Propogate errors
339 types::Type* calleeType = _ctx.types.table.getNodeType(node.getCallee());
340 if (calleeType == errorType) {
341 // ^^ Note that this wont break if the callee is a function set since it
342 // will actually be nullptr instead
343 _ctx.types.table.setNodeType(&node, errorType);
344 return;
345 }
346
347 // Handle function sets specially, since we can only decide which overload here when
348 // we have argument context
349 ast::NameExprNode* calleeNameExpr = ast::ast_cast<ast::NameExprNode>(node.getCallee());
350 if (calleeNameExpr) {
351 symbols::Symbol* resolvedSymbol = calleeNameExpr->getResolvedSymbol();
352 if (!resolvedSymbol) {
353 // Cannot infer unresolved symbol
354 _ctx.types.table.setNodeType(&node, errorType);
355 return;
356 }
357
358 symbols::FunctionSetSymbol* funcSet = resolvedSymbol->as<symbols::FunctionSetSymbol>();
359 if (!funcSet) {
360 // Cannot call non-function set symbols
361 _ctx.diagnostics.report(
362 diagnostics::ERROR_CALL_NON_FUNCTION,
363 node.getRange(),
364 _ctx.strings.get(resolvedSymbol->getNameValue())
365 );
366
367 _ctx.types.table.setNodeType(&node, errorType);
368 return;
369 }
370
371 // Grab argument types
372 std::vector<types::Type*> argTypes;
373 for (ast::ExpressionNode* arg : node.getArgs()) {
374 types::Type* argType = _ctx.types.table.getNodeType(arg);
375 VEE_ASSERT(argType != nullptr, "Failed to infer type for argument of call expression");
376 argTypes.push_back(argType);
377 }
378
379 // Lookup overloads
380 std::vector<symbols::FunctionSymbol*> overloads = lookupOverload(funcSet, argTypes);
381
382 if (overloads.empty()) {
383 // No matching overload found
384 _ctx.diagnostics.report(
385 diagnostics::ERROR_FUNCTION_OVERLOAD_NOT_FOUND,
386 node.getRange(),
387 _ctx.strings.get(funcSet->getNameValue())
388 );
389
390 _ctx.types.table.setNodeType(calleeNameExpr, errorType);
391 _ctx.types.table.setNodeType(&node, errorType);
392 return;
393 } else if (overloads.size() > 1) {
394 // Ambiguous overload
395 _ctx.diagnostics.report(
396 diagnostics::ERROR_FUNCTION_OVERLOAD_AMBIGUOUS,
397 node.getRange(),
398 _ctx.strings.get(funcSet->getNameValue())
399 );
400
401 _ctx.types.table.setNodeType(calleeNameExpr, errorType);
402 _ctx.types.table.setNodeType(&node, errorType);
403 return;
404 } else {
405 // Found exactly one matching overload
406 // IMPORTANT: Use function type for name expression node,
407 // and return type for call expression node
408 symbols::FunctionSymbol* funcSymbol = overloads[0];
409 types::FunctionType* funcType = funcSymbol->getType()->as<types::FunctionType>();
410 VEE_ASSERT(funcType != nullptr, "Function symbol does not have a function type");
411
412 // Store in node
413 node.symbol = funcSymbol;
414
415 _ctx.types.table.setNodeType(calleeNameExpr, funcType);
416 _ctx.types.table.setNodeType(&node, funcType->getReturnType());
417
418 // Output conversion diagnostics for args
419 const std::vector<types::Type*>& paramTypes = funcType->getParameterTypes();
420 for (size_t i = 0; i < argTypes.size(); ++i) {
421 types::Type* argType = argTypes[i];
422 types::Type* paramType = paramTypes[i];
423
424 emitImplicitConversionDiagnostics(
425 node.getArgs()[i]->getRange(),
426 argType,
427 paramType
428 );
429 }
430
431 return;
432 }
433 }
434
435 // Any function-typed object
436 types::FunctionType* calleeTypeAsFunction = calleeType->as<types::FunctionType>();
437 if (calleeTypeAsFunction) {
438 // Result type is the return type of the function
439 types::Type* resultType = calleeTypeAsFunction->getReturnType();
440 _ctx.types.table.setNodeType(&node, resultType);
441 return;
442 }
443
444 // Not a function (or object with function type)
445 _ctx.types.table.setNodeType(&node, errorType);
446}
447void TypeCheckerPass::visitIndexExpr(ast::IndexExprNode& node) {
448 ast::AstWalker::visitIndexExpr(node);
449
450 types::Type* objectType = _ctx.types.table.getNodeType(node.getObject());
451 types::Type* indexType = _ctx.types.table.getNodeType(node.getIndex());
452
453 types::ErrorType* errorType = _ctx.types.table.getError();
454
455 // Propogate errors
456 if (objectType == errorType || indexType == errorType) {
457 _ctx.types.table.setNodeType(&node, errorType);
458 return;
459 }
460
461 // For now, only arrays can be indexed
462 // maybe in the future pointers or other types can be indexed too
463 types::ArrayType* objectTypeAsArray = objectType->as<types::ArrayType>();
464 if (objectTypeAsArray) {
465 // Index must be an integer type
466 types::BuiltinType* indexTypeAsBuiltin = indexType->as<types::BuiltinType>();
467 if (indexTypeAsBuiltin && indexTypeAsBuiltin->isInteger()) {
468 // Result type is the element type of the array
469 types::Type* resultType = objectTypeAsArray->getElementType();
470 _ctx.types.table.setNodeType(&node, resultType);
471 return;
472 }
473 }
474
475 // TODO: Support index operator overloads (perhaps based on index type too?)
476 _ctx.types.table.setNodeType(&node, errorType);
477}

527void TypeCheckerPass::visitVariableDecl(ast::VariableDeclNode& node) {
528 ast::AstWalker::visitVariableDecl(node);
529 // ^^ Walks type and initializer
530
531 types::ErrorType* errorType = _ctx.types.table.getError();
532
533 // NOTE: We dont want to use setNodeType here, since its a declaration,
534 // instead, only set the type of the symbol vv
535 symbols::VariableSymbol* varSymbol = node.symbol;
536
537 // If we have no initializer we dont need to do any checks
538 // UNLESS the variable has no explicit type
539 if (node.getInitializer() == nullptr) {
540 if (varSymbol->getType() == nullptr) {
541 // Cannot infer type for variable without type or initializer
542 _ctx.diagnostics.report(
543 diagnostics::ERROR_VAR_DECL_NO_TYPE_OR_INIT,
544 node.getRange(),
545 _ctx.strings.get(node.getName().id)
546 );
547
548 varSymbol->setType(_ctx.types.table.getError());
549 }
550 return;
551 }
552
553 types::Type* initType = _ctx.types.table.getNodeType(node.getInitializer());
554 VEE_ASSERT(initType != nullptr, "Failed to infer type for initializer of variable declaration");
555
556 // Attempt to infer type if not explicitly specified
557 if (varSymbol->getType() == nullptr) {
558 varSymbol->setType(initType);
559
560 // If we inferred an error type, return early to avoid further errors
561 if (initType == errorType) {
562 return;
563 }
564 }
565
566 // Validate initializer can be assigned to variable
567 if (!implicitConversionPossible(initType, varSymbol->getType())) {
568 // Cannot convert initializer type to variable type
569 _ctx.diagnostics.report(
570 diagnostics::ERROR_VAR_ASSIGNMENT_TYPE_MISMATCH,
571 node.getInitializer()->getRange(),
572 initType->toString(),
573 varSymbol->getType()->toString()
574 );
575 }
576
577 // Emit conversion diagnostics for initializer
578 emitImplicitConversionDiagnostics(
579 node.getInitializer()->getRange(),
580 initType,
581 varSymbol->getType()
582 );
583}

592template<typename T, typename CostFn>
593std::vector<T*> TypeCheckerPass::findBestCandidates(std::span<T* const> candidates, CostFn&& costFn) {
594 std::vector<T*> results;
595
596 u32 lowestCost = types::kNoConversionCost;
597 for (const auto& candidate : candidates) {
598 u32 cost = costFn(candidate);
599
600 if (cost < lowestCost) {
601 lowestCost = cost;
602 results.clear();
603 results.push_back(candidate);
604 }
605 else if (cost == lowestCost) {
606 results.push_back(candidate);
607 }
608 }
609
610 return results;
611}

676std::vector<symbols::OperatorSymbol*> TypeCheckerPass::lookupUnaryOperator(symbols::UnaryOperatorKind kind, types::Type* operandType) {
677 VEE_ASSERT(operandType != nullptr, "Operand type must not be null");
678
679 // Early out if error type is passed
680 if (operandType == _ctx.types.table.getError()) {
681 return {};
682 }
683
684 std::span<symbols::OperatorSymbol* const> candidates = _ctx.sema.operators.getUnaryOperators(kind);
685
686 return findBestCandidates(candidates, [&](symbols::OperatorSymbol* opSymbol) {
687 types::Type* opType = opSymbol->getOperandType(0);
688 return implicitConversionCost(operandType, opType);
689 });
690}
691std::vector<symbols::OperatorSymbol*> TypeCheckerPass::lookupBinaryOperator(symbols::BinaryOperatorKind kind, types::Type* leftType, types::Type* rightType) {
692 VEE_ASSERT(leftType != nullptr, "Left operand type must not be null");
693 VEE_ASSERT(rightType != nullptr, "Right operand type must not be null");
694
695 // Early out if error type is passed
696 types::Type* errorType = _ctx.types.table.getError();
697 if (leftType == errorType || rightType == errorType) {
698 return {};
699 }
700
701 std::span<symbols::OperatorSymbol* const> candidates = _ctx.sema.operators.getBinaryOperators(kind);
702
703 return findBestCandidates(candidates, [&](symbols::OperatorSymbol* opSymbol) {
704 // Left cost
705 types::Type* leftOpType = opSymbol->getOperandType(0);
706 u32 leftCost = implicitConversionCost(leftType, leftOpType);
707 if (leftCost == types::kNoConversionCost)
708 return types::kNoConversionCost;
709
710 // Right cost
711 types::Type* rightOpType = opSymbol->getOperandType(1);
712 u32 rightCost = implicitConversionCost(rightType, rightOpType);
713 if (rightCost == types::kNoConversionCost)
714 return types::kNoConversionCost;
715
716 // Combined cost is sum of left and right costs
717 return leftCost + rightCost;
718 });
719}
720std::vector<symbols::FunctionSymbol*> TypeCheckerPass::lookupOverload(symbols::FunctionSetSymbol* set, const std::vector<types::Type*>& argTypes) {
721 VEE_ASSERT(set != nullptr, "Function set symbol must not be null");
722 for (const auto* argType : argTypes) {
723 VEE_ASSERT(argType != nullptr, "Argument type must not be null");
724 }
725
726 // Early out if error type is passed
727 types::Type* errorType = _ctx.types.table.getError();
728 if (std::any_of(argTypes.begin(), argTypes.end(), [&](types::Type* t) { return t == errorType; })) {
729 return {};
730 }
731
732 std::span<symbols::FunctionSymbol* const> candidates = set->getOverloads();
733
734 return findBestCandidates(candidates, [&](symbols::FunctionSymbol* funcSymbol) {
735 types::FunctionType* funcType = funcSymbol->getType()->as<types::FunctionType>();
736 VEE_ASSERT(funcType != nullptr, "Function symbol must have a function type");
737
738 const auto& paramTypes = funcType->getParameterTypes();
739 if (paramTypes.size() != argTypes.size()) {
740 return types::kNoConversionCost;
741 }
742
743 u32 totalCost = 0;
744 for (size_t i = 0; i < paramTypes.size(); ++i) {
745 u32 cost = implicitConversionCost(argTypes[i], paramTypes[i]);
746 if (cost == types::kNoConversionCost) {
747 return types::kNoConversionCost;
748 }
749 totalCost += cost;
750 }
751
752 return totalCost;
753 });
754}

756void TypeCheckerPass::emitImplicitConversionDiagnostics(
757 const source::SourceRange& range,
758 types::Type* fromType,
759 types::Type* toType
760) {
761 using ConversionRank = types::ConversionRank;
762
763 VEE_ASSERT(fromType != nullptr, "From type must not be null");
764 VEE_ASSERT(toType != nullptr, "To type must not be null");
765
766 // Early out no conversion
767 if (fromType == toType) {
768 return;
769 }
770
771 ConversionRank rank = _ctx.types.system.rankConversion(fromType, toType, types::ConversionMode::Implicit);
772 switch (rank) {
773 case ConversionRank::ExactMatch:
774 case ConversionRank::Promotion:
775 case ConversionRank::Conversion:
776 case ConversionRank::UserDefinedConversion:
777 case ConversionRank::ExplicitConversion:
778 // No diagnostics for these ranks
779 break;
780
781 case ConversionRank::NarrowingConversion:
782 _ctx.diagnostics.report(
783 diagnostics::WARNING_NARROWING_CONVERSION,
784 range,
785 fromType->toString(),
786 toType->toString()
787 );
788 break;
789
790 case ConversionRank::NoConversion:
791 _ctx.diagnostics.report(
792 diagnostics::ERROR_NO_IMPLICIT_CONVERSION_AVAILABLE,
793 range,
794 fromType->toString(),
795 toType->toString()
796 );
797 break;
798
799 default:
800 VEE_UNREACHABLE("Unknown conversion rank");
801 }
802}
