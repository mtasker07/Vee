65void TypeCheckerPass::visitIntLiteralExpr(ast::IntLiteralExprNode& node) {
66 using BTK = types::BuiltinTypeKind;
67
68 ast::AstWalker::visitIntLiteralExpr(node);
69
70 // Integer literals are always i32
71 // TODO: Support suffixes for different literal types
72 types::BuiltinType* intType = _sema.types.getBuiltin(types::BuiltinTypeKind::I32);
73 _sema.types.setNodeType(&node, intType);
74
75 // Validate size of integer literal
76 const basic::BigInt& value = node.getValue();
77
78 u32 bitWidth = 0;
79 bool isSigned = false;
80 BTK builtinKind = intType->getBuiltinKind();
81 switch (builtinKind) {
82 case BTK::I8:
83 case BTK::U8:
84 bitWidth = 8;
85 isSigned = (builtinKind == BTK::I8);
86 break;
87 case BTK::I16:
88 case BTK::U16:
89 bitWidth = 16;
90 isSigned = (builtinKind == BTK::I16);
91 break;
92 case BTK::I32:
93 case BTK::U32:
94 bitWidth = 32;
95 isSigned = (builtinKind == BTK::I32);
96 break;
97 case BTK::I64:
98 case BTK::U64:
99 bitWidth = 64;
100 isSigned = (builtinKind == BTK::I64);
101 break;
102 }
103
104 if (isSigned) {
105 if (!value.fitsInSigned(bitWidth)) {
106 _ctx.diagnostics.report(diagnostics::ERROR_INT_LITERAL_OUT_OF_RANGE, node.getRange(), value.toString(), bitWidth);
107 }
108 } else {
109 if (!value.fitsInUnsigned(bitWidth)) {
110 _ctx.diagnostics.report(diagnostics::ERROR_INT_LITERAL_OUT_OF_RANGE, node.getRange(), value.toString(), bitWidth);
111 }
112 }
113}

253void TypeCheckerPass::visitIndexExpr(ast::IndexExprNode& node) {
254 ast::AstWalker::visitIndexExpr(node);
255
256 types::Type* objectType = _sema.types.getNodeType(node.getObject());
257 types::Type* indexType = _sema.types.getNodeType(node.getIndex());
258
259 types::ErrorType* errorType = _sema.types.getError();
260
261 // Propogate errors
262 if (objectType == errorType || indexType == errorType) {
263 _sema.types.setNodeType(&node, errorType);
264 return;
265 }
266
267 // For now, only arrays can be indexed
268 // maybe in the future pointers or other types can be indexed too
269 types::ArrayType* objectTypeAsArray = objectType->as<types::ArrayType>();
270 if (objectTypeAsArray) {
271 // Index must be an integer type
272 types::BuiltinType* indexTypeAsBuiltin = indexType->as<types::BuiltinType>();
273 if (indexTypeAsBuiltin && indexTypeAsBuiltin->isInteger()) {
274 // Result type is the element type of the array
275 types::Type* resultType = objectTypeAsArray->getElementType();
276 _sema.types.setNodeType(&node, resultType);
277 return;
278 }
279 }
280
281 // TODO: Support index operator overloads (perhaps based on index type too?)
282 _sema.types.setNodeType(&node, errorType);
283}
284void TypeCheckerPass::visitMemberAccessExpr(ast::MemberAccessExprNode& node) {
285 ast::AstWalker::visitMemberAccessExpr(node);
286
287 types::Type* objectType = _sema.types.getNodeType(node.getObject());
288
289 types::ErrorType* errorType = _sema.types.getError();
290
291 // Propogate errors
292 if (objectType == errorType) {
293 _sema.types.setNodeType(&node, errorType);
294 return;
295 }
296
297 types::ClassType* objectTypeAsClass = objectType->as<types::ClassType>();
298 if (objectTypeAsClass) {
299 // Look up member
300 types::Type* memberType = errorType;
301
302 if (const auto* fieldSymbol = objectTypeAsClass->getField(node.getMember().id)) {
303 memberType = fieldSymbol->getType();
304 }
305 else if (const auto* methodSymbols = objectTypeAsClass->getMethods(node.getMember().id)) {
306 // TODO: Identify overload
307 memberType = methodSymbols->front()->getType();
308 }
309
310 _sema.types.setNodeType(&node, memberType);
311 return;
312 }
