53void SymbolCollectionPass::visitModuleDecl(ast::ModuleDeclNode& node) {
54 symbols::ModuleSymbol* oldCurrentModule = _currentModule;
55
56 basic::StringId modName = node.getName().id;
57
58 // Declare symbol
59 symbols::ModuleSymbol* modSym = _sema.symbols.declare<symbols::ModuleSymbol>(modName);
60 VEE_ASSERT(modSym != nullptr, "Failed to declare module symbol: {}", _ctx.strings.get(modName));
61
62 // Resolve node to symbol
63 node.symbol = modSym;
64
65 // Bind module name -> module
66 if (existsBindingNamed(modName)) {
67 reportNameAlreadyDeclaredInScope(node, "module", modName);
68 }
69 else {
70 ensureBind(modName, modSym);
71 }
72
73 _currentModule = modSym;
74
75 // Create scope for module
76 sema::Scope* moduleScope = _sema.scopes.createScope(currentScope());
77 _sema.scopes.setNodeScope(&node, moduleScope);
78 modSym->bindScope(moduleScope);
79 pushScope(moduleScope);
80
81 // Walk node normally
82 ast::AstWalker::visitModuleDecl(node);
83
84 popScope();
85 _currentModule = oldCurrentModule;

87void SymbolCollectionPass::visitFunctionDecl(ast::FunctionDeclNode& node) {
88 symbols::FunctionSymbol* oldCurrentFunctionOrMethod = _currentFunctionOrMethod;
89
90 basic::StringId funcName = node.getName().id;
91
92 // Declare symbol
93 auto funcSym = _sema.symbols.declare<symbols::FunctionSymbol>(funcName);
94 VEE_ASSERT(funcSym != nullptr, "Failed to declare function symbol: {}", _ctx.strings.get(funcName));
95
96 // Resolve node to symbol
97 node.symbol = funcSym;
98
99 // Bind function name -> function set symbol
100 if (symbols::Symbol* existingSym = bindingNamed(funcName)) {
101 // Existing set -> add overload
102 if (existingSym->is<symbols::FunctionSetSymbol>()) {
103 symbols::FunctionSetSymbol* funcSetSym = existingSym->as<symbols::FunctionSetSymbol>();
104 funcSetSym->addOverload(funcSym);
105 }
106 // Existing non-function -> report
107 else {
108 reportNameAlreadyDeclaredInScope(node, "function", funcName);
109 }
110 }
111 else {
112 // New set
113 auto funcSetSym = _sema.symbols.declare<symbols::FunctionSetSymbol>(funcName);
114 VEE_ASSERT(funcSetSym != nullptr, "Failed to declare function set symbol: {}", _ctx.strings.get(funcName));
115
116 // Add initial overload
117 funcSetSym->addOverload(funcSym);
118
119 // Bind by name
120 ensureBind(funcName, funcSetSym);
121 }
122
123 _currentFunctionOrMethod = funcSym;
124
125 // Create scope for function
126 sema::Scope* funcScope = _sema.scopes.createScope(currentScope());
127 _sema.scopes.setNodeScope(&node, funcScope);
128 funcSym->bindScope(funcScope);
129 pushScope(funcScope);
130
131 ast::AstWalker::visitFunctionDecl(node);
132
133 popScope();
134 _currentFunctionOrMethod = oldCurrentFunctionOrMethod;

304 VEE_ASSERT(currentScope() != nullptr, "No current scope available");

309void SymbolCollectionPass::ensureBind(basic::StringId nameId, symbols::Symbol* symbol) {
310 VEE_ASSERT(currentScope() != nullptr, "No current scope available");
311 symbols::IdentifierTable& idTable = currentScope()->getIdentifierTable();
312 VEE_ASSERT(idTable.bind(nameId, symbol),
313 "Failed to bind name to symbol: {}", _ctx.strings.get(nameId));

316void SymbolCollectionPass::reportNameAlreadyDeclaredInScope(const ast::AstNode& node, std::string_view symbolKind, basic::StringId nameId) {
317 symbols::Symbol* existingSym = currentScope()->getIdentifierTable().lookup(nameId);
318 _ctx.diagnostics.report(

320 node.getRange(),
321 symbolKind,
322 _ctx.strings.get(nameId),
