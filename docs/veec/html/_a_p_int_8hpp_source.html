185template<>
186struct hash<veec::basic::APInt> {
187 size_t operator()(const veec::basic::APInt& key) const {
188 u64 h = 0x9e3779b97f4a7c15ULL;
189
190 for (u32 limb : key._words) {
191 h ^= std::rotl(static_cast<u64>(limb) * 0xbf58476d1ce4e5b9ULL, 27);
192 h *= 0x94d049bb133111ebULL;
193 }
194
195 if (key.isNegative())
196 h ^= 0xdeadbeefcafebabeULL;
197
198 return static_cast<size_t>(h);
