371Value* MirBuilder::createTruncateInt(Value* value, types::Type* toType, std::string_view resultName) {
372 assertInteger(value, "value");
373 VEE_ASSERT(toType != nullptr, "toType is null!");
374
375 types::BuiltinType* builtinValueType = _mir.valueTypes.getValueType(value)->as<types::BuiltinType>();
376 u32 valueBitWidth = builtinValueType->getBitWidth();
377
378 types::BuiltinType* builtinToType = toType->as<types::BuiltinType>();
379 VEE_ASSERT(builtinToType->isInteger(), "toType must be an integer type!");
380 u32 toBitWidth = builtinToType->getBitWidth();
381
382 VEE_ASSERT(toBitWidth < valueBitWidth, "toType must have a smaller bit width than the value type!");
383 VEE_ASSERT(builtinValueType->isSignedInteger() == builtinToType->isSignedInteger(),
384 "Invalid integer truncation: from {} to {}. Both types must be of the same signedness",
385 builtinValueType->toString(), builtinToType->toString());
386
387 return createValueInstruction(InstructionOpcode::TruncateInt, toType, { value }, resultName);
388}
389Value* MirBuilder::createExtendInt(Value* value, types::Type* toType, std::string_view resultName) {
390 assertInteger(value, "value");
391 VEE_ASSERT(toType != nullptr, "toType is null!");
392
393 types::BuiltinType* builtinValueType = _mir.valueTypes.getValueType(value)->as<types::BuiltinType>();
394 u32 valueBitWidth = builtinValueType->getBitWidth();
395
396 types::BuiltinType* builtinToType = toType->as<types::BuiltinType>();
397 VEE_ASSERT(builtinToType->isInteger(), "toType must be an integer type!");
398 u32 toBitWidth = builtinToType->getBitWidth();
399
400 VEE_ASSERT(toBitWidth > valueBitWidth, "toType must have a larger bit width than the value type!");
401
402 if (builtinValueType->isSignedInteger() && builtinToType->isSignedInteger()) {
403 return createValueInstruction(InstructionOpcode::SignExtendInt, toType, { value }, resultName);
404 } else if (builtinValueType->isUnsignedInteger() && builtinToType->isUnsignedInteger()) {
405 return createValueInstruction(InstructionOpcode::ZeroExtendInt, toType, { value }, resultName);
406 }
407
408 VEE_FATAL("Invalid integer extension: from {} to {}. Both types must be of the same signedness",
409 builtinValueType->toString(), builtinToType->toString());
410}

486 VEE_ASSERT(function != nullptr, "Function is null!");

497 return createValueInstruction(InstructionOpcode::Call, returnType, std::move(operands), resultName);
