41void TypeResolutionPass::visitFunctionDecl(ast::FunctionDeclNode& node) {
42 ast::AstWalker::visitFunctionDecl(node);
43 // ^^ Assigns parameter types and return type
44
45 symbols::FunctionSymbol* functionSymbol = node.symbol;
46 VEE_ASSERT(functionSymbol != nullptr, "FunctionDeclNode has no associated FunctionSymbol");
47
48 // Set the return type and parameters
49 types::Type* returnType = _ctx.types.table.getNodeType(node.getReturnType());
50
51 std::vector<types::Type*> parameterTypes;
52 for (ast::ParameterDeclNode* paramNode : node.getParameters()) {
53 // Add to symbol
54 functionSymbol->addParameter(paramNode->symbol);
55 parameterTypes.push_back(_ctx.types.table.getNodeType(paramNode->getType()));
56 }
57
58 // Assign to symbol
59 types::FunctionType* funcType = _ctx.types.table.getFunction(returnType, parameterTypes);
60 functionSymbol->setType(funcType);
61}

127void TypeResolutionPass::visitBuiltinType(ast::BuiltinTypeNode& node) {
128 ast::AstWalker::visitBuiltinType(node);
129
130 // Convert to builtin type (from ast builtin type)
131 types::BuiltinTypeKind kind;
132 switch (node.getBuiltinTypeKind()) {
133 case ast::BuiltinTypeKind::Void: kind = types::BuiltinTypeKind::Void; break;
134 case ast::BuiltinTypeKind::Bool: kind = types::BuiltinTypeKind::Bool; break;
135 case ast::BuiltinTypeKind::String: kind = types::BuiltinTypeKind::String; break;
136 case ast::BuiltinTypeKind::I8: kind = types::BuiltinTypeKind::I8; break;
137 case ast::BuiltinTypeKind::I16: kind = types::BuiltinTypeKind::I16; break;
138 case ast::BuiltinTypeKind::I32: kind = types::BuiltinTypeKind::I32; break;
139 case ast::BuiltinTypeKind::I64: kind = types::BuiltinTypeKind::I64; break;
140 case ast::BuiltinTypeKind::U8: kind = types::BuiltinTypeKind::U8; break;
141 case ast::BuiltinTypeKind::U16: kind = types::BuiltinTypeKind::U16; break;
142 case ast::BuiltinTypeKind::U32: kind = types::BuiltinTypeKind::U32; break;
143 case ast::BuiltinTypeKind::U64: kind = types::BuiltinTypeKind::U64; break;
144 case ast::BuiltinTypeKind::F32: kind = types::BuiltinTypeKind::F32; break;
145 case ast::BuiltinTypeKind::F64: kind = types::BuiltinTypeKind::F64; break;
146 default:
147 VEE_UNREACHABLE("Unknown builtin type kind");
148 }
149
150 types::Type* type = _ctx.types.table.getBuiltin(kind);
151 _ctx.types.table.setNodeType(&node, type);
152}
153void TypeResolutionPass::visitNamedType(ast::NamedTypeNode& node) {
154 ast::AstWalker::visitNamedType(node);
155
156 types::ErrorType* errorType = _ctx.types.table.getError();
157
158 symbols::Symbol* resolvedSymbol = node.getResolvedSymbol();
159 if (!resolvedSymbol || !resolvedSymbol->is<symbols::ClassSymbol>()) {
160 _ctx.types.table.setNodeType(&node, errorType);
161 return;
162 }
163
164 symbols::ClassSymbol* classSymbol = resolvedSymbol->as<symbols::ClassSymbol>();
165
166 types::Type* classType = _ctx.types.table.getClass(classSymbol);
167 _ctx.types.table.setNodeType(&node, classType);
168}
