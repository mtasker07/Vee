36std::string MirPrinter::printNode(const MirNode& node) {
37 _oss.str("");
38 _oss.clear();
39 _indent = 0;
40
41 switch (node.getNodeKind()) {
42 case MirKind::Module:
43 printModule(static_cast<const Module&>(node));
44 break;
45 case MirKind::Function:
46 printFunction(static_cast<const Function&>(node));
47 break;
48 case MirKind::BasicBlock:
49 printBasicBlock(static_cast<const BasicBlock&>(node));
50 break;
51 case MirKind::Instruction:
52 printInstruction(static_cast<const Instruction&>(node));
53 break;
54 case MirKind::Local:
55 case MirKind::Constant:
56 printOperand(static_cast<const Value&>(node));
57 break;
58 default:
59 VEE_UNREACHABLE("Unknown MIR node kind");
60 }
61 return _oss.str();
62}

80void MirPrinter::printFunction(const Function& function) {
81 // Values are unique to functions, so reset the counter here
82 _valueCounter = 1;
83 _unnamedValueNames.clear();
84
85 // Print signature
86 std::string_view funcName = nameOfFunction(function);
87 std::string args;
88 for (size_t i = 0; i < function.getParameterCount(); ++i) {
89 const Local& param = *function.getParameter(i);
90 if (i > 0) args += ", ";
91 std::string paramTypeName = typeOfValue(param)->toString();
92 std::string_view paramName = nameOfValue(param);
93 args += std::format("{} {}", paramTypeName, paramName);
94 }
95 std::string returnTypeName = typeOfValue(function)->toString();
96
97 // Print body
98 addLineIndented(std::format("func {}({}) -> {} {{", funcName, args, returnTypeName));
99 _indent++;
100 for (const BasicBlock* block : function.getBlocks()) {
101 printBasicBlock(*block);
102 }
103 _indent--;
104 addLineIndented("}");
105}

114void MirPrinter::printInstruction(const Instruction& instruction) {
115 std::ostringstream line;
116
117 // Print result value if instruction produces one
118 if (instruction.producesValue()) {
119 std::string_view valueName = nameOfValue(instruction);
120 line << std::format("%{} = ", valueName);
121 }
122
123 // Print opcode
124 std::string_view opcodeStr = toString(instruction.getOpcode());
125 line << opcodeStr;
126
127 // Print operands
128 for (size_t i = 0; i < instruction.getOperandCount(); ++i) {
129 line << " ";
130 line << printOperand(*instruction.getOperand(i));
131 if (i < instruction.getOperandCount() - 1) {
132 line << ",";
133 }
134 }
135
136 addLineIndented(line.str());
137}
138std::string MirPrinter::printOperand(const Value& value) {
139 switch (value.getNodeKind()) {
140 case MirKind::Function: {
141 const Function* func = static_cast<const Function*>(&value);
142 return std::format("@{}", nameOfFunction(*func));
143 }
144 case MirKind::BasicBlock: {
145 const BasicBlock* block = static_cast<const BasicBlock*>(&value);
146 return std::format("@{}", nameOfValue(*block));
147 }
148 case MirKind::Instruction: {
149 const Instruction* instr = static_cast<const Instruction*>(&value);
150 return std::format("%{}", nameOfValue(*instr));
151 }
152 case MirKind::Local: {
153 const Local* local = static_cast<const Local*>(&value);
154 std::string typeName = typeOfValue(*local)->toString();
155 return std::format("{} %{}", typeName, nameOfValue(*local));
156 }
157 case MirKind::Constant: {
158 const Constant* constant = static_cast<const Constant*>(&value);
159 std::string typeName = typeOfValue(*constant)->toString();
160 return std::format("{} ${}", typeName, nameOfValue(*constant));
161 }
162
163 default:
164 VEE_UNREACHABLE("Unknown Value kind");
165 }

167std::string MirPrinter::printConstant(const Constant& constant) {
168 std::string_view constName = _ctx.mir.valueNames.getName(&constant);
169
170 std::string valueStr;
171 switch (constant.getKind()) {
172 case ConstantKind::Int: {
173 const ConstantInt* intConst = constant.asInteger();
174 valueStr = intConst->getValue().toString();
175 break;
176 }
177 case ConstantKind::Float: {
178 const ConstantFloat* floatConst = constant.asFloat();
179 valueStr = std::to_string(floatConst->getValue());
180 break;
181 }
182 case ConstantKind::String: {
183 const ConstantString* stringConst = constant.asString();
184 valueStr = std::string(stringConst->getValue());
185 break;
186 }
187 case ConstantKind::Bool: {
188 const ConstantBool* boolConst = constant.asBool();
189 valueStr = boolConst->getValue() ? "true" : "false";
190 break;
191 }
192 }
193
194 return std::format("${} = {}", constName, valueStr);
195}

216std::string_view MirPrinter::nameOfValue(const Value& value) const {
217 std::string_view name = _ctx.mir.valueNames.getName(&value);
218 if (name.empty()) {
219 if (!_unnamedValueNames.contains(&value)) {
220 _unnamedValueNames[&value] = std::to_string(_valueCounter++);
221 }
222 return _unnamedValueNames[&value];
223 }
224 return name;

233 }
