53void TypeInferencePass::visitBuiltinType(ast::BuiltinTypeNode& node) {
54 ast::AstWalker::visitBuiltinType(node);
55
56 // Convert to builtin type (from ast builtin type)
57 types::BuiltinTypeKind kind;
58 switch (node.getBuiltinTypeKind()) {
59 case ast::BuiltinTypeKind::Void: kind = types::BuiltinTypeKind::Void; break;
60 case ast::BuiltinTypeKind::Bool: kind = types::BuiltinTypeKind::Bool; break;
61 case ast::BuiltinTypeKind::String: kind = types::BuiltinTypeKind::String; break;
62 case ast::BuiltinTypeKind::I8: kind = types::BuiltinTypeKind::I8; break;
63 case ast::BuiltinTypeKind::I16: kind = types::BuiltinTypeKind::I16; break;
64 case ast::BuiltinTypeKind::I32: kind = types::BuiltinTypeKind::I32; break;
65 case ast::BuiltinTypeKind::I64: kind = types::BuiltinTypeKind::I64; break;
66 case ast::BuiltinTypeKind::U8: kind = types::BuiltinTypeKind::U8; break;
67 case ast::BuiltinTypeKind::U16: kind = types::BuiltinTypeKind::U16; break;
68 case ast::BuiltinTypeKind::U32: kind = types::BuiltinTypeKind::U32; break;
69 case ast::BuiltinTypeKind::U64: kind = types::BuiltinTypeKind::U64; break;
70 case ast::BuiltinTypeKind::F32: kind = types::BuiltinTypeKind::F32; break;
71 case ast::BuiltinTypeKind::F64: kind = types::BuiltinTypeKind::F64; break;
72 default:
73 VEE_UNREACHABLE("Unknown builtin type kind");
74 }
75
76 types::Type* type = _sema.types.getBuiltin(kind);
77 _sema.types.setNodeType(&node, type);
78}
79void TypeInferencePass::visitNamedType(ast::NamedTypeNode& node) {
80 ast::AstWalker::visitNamedType(node);
81
82 types::ErrorType* errorType = _sema.types.getError();
83
84 symbols::Symbol* resolvedSymbol = node.getResolvedSymbol();
85 if (!resolvedSymbol || !resolvedSymbol->is<symbols::ClassSymbol>()) {
86 _sema.types.setNodeType(&node, errorType);
87 return;
88 }
89
90 symbols::ClassSymbol* classSymbol = resolvedSymbol->as<symbols::ClassSymbol>();
91
92 types::Type* classType = _sema.types.getClass(classSymbol);
93 _sema.types.setNodeType(&node, classType);
94}
