471 // Allocate heap memory for newCapacity elements and set _data to point to it
472 inline void allocateHeap(size_t newCapacity) {
473 _data = AllocatorTraits::allocate(_allocator, newCapacity);
474 _capacity = newCapacity;

476 // Allocate heap memory for newCapacity elements, move existing elements
477 // into it, and deallocate the old memory
478 inline void reallocateHeap(size_t newCapacity) {
479 T* newData = AllocatorTraits::allocate(_allocator, newCapacity);
480
481 moveConstructRange(_data, newData, _size);
482 destroyRange(_data, _size);
483 deallocateHeap();
484
485 _data = newData;
486 _capacity = newCapacity;
487 }
488 // Move elements to inline storage and deallocate heap memory
489 inline void moveToInline() {
490 if (is_inline())
491 return;
492
493 T* newData = inline_data();
494
495 moveConstructRange(_data, newData, _size);
496 destroyRange(_data, _size);
497 deallocateHeap();
498
499 _data = newData;
500 _capacity = InlineCapacity;
501 }

503 // Construct `count` elements from `src` into `dest` using move semantics
504 inline void moveConstructRange(T* src, T* dest, size_t count) noexcept {
505 for (size_t i = 0; i < count; ++i)
506 AllocatorTraits::construct(_allocator, &dest[i], std::move(src[i]));
507 }
508 // Construct `count` elements from `src` into `dest` using copy semantics
509 inline void copyConstructRange(T* src, T* dest, size_t count) noexcept {
510 for (size_t i = 0; i < count; ++i)
511 AllocatorTraits::construct(_allocator, &dest[i], src[i]);

529template<typename T, size_t N, class Allocator>
530inline void SmallVector<T, N, Allocator>::copyFrom(const SmallVector& other) {
531 destroyRange(_data, _size);
532
533 if (other._size > _capacity) {
534 deallocateHeap();
535 allocateHeap(other._capacity);
536 }
537
538 copyConstructRange(other._data, _data, other._size);
539 _size = other._size;
540}
541template<typename T, size_t N, class Allocator>
542inline void SmallVector<T, N, Allocator>::moveFrom(SmallVector&& other) {
543 // O(N) if other is inline, O(1) if other is heap allocated
544 if (other.is_inline()) {
545 if (other._size > _capacity) {
546 deallocateHeap();
547 allocateHeap(other._capacity);
548 }
549
550 moveConstructRange(other._data, _data, other._size);
551 destroyRange(other._data, other._size);
552 }
553 else {
554 destroyRange(_data, _size);
555 deallocateHeap();
556
557 _data = other._data;
558 _capacity = other._capacity;
559 }
560 _size = other._size;
561
562 other._data = other.inline_data();
563 other._capacity = InlineCapacity;
564 other._size = 0;
565}
